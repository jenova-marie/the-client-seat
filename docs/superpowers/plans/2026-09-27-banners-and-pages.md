> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Rename the series to The Client Seat, render twelve branded 1600×900 banner PNGs from an HTML template over the author's background art, publish them at GitHub Pages URLs, and make the Medium render script inject each banner above its post.

**Architecture:** A Python generator reads the twelve post H1s into a manifest, fills one HTML/CSS template per part, and screenshots each with the Chrome already installed on this Mac. A second Python script is the test for the banners. The existing bash render script gains an image injection and a corrected pandoc flag. Pages serves the repo root so the committed PNGs get stable URLs.

**Tech Stack:** Python 3.12 with Pillow (installed), Google Chrome headless (installed at the standard macOS path), pandoc 3.8, git, gh (signed in as jenova-marie). Fonts: Fraunces and Inter, Open Font License, vendored from the google/fonts repository.

**Spec:** `docs/superpowers/specs/2026-09-27-banners-and-pages-design.md`

- Banner canvas exactly 1600 × 900 PNG; text block on the left third, not inside the outer 80 px. (spec §5)
- Banner text is only: `THE CLIENT SEAT`, `Part N`, the part title, `Jenova`. No client detail. (spec §3)
- Hue shift per part: `(part × 9)` degrees, low-opacity overlay. (spec §5)
- Fonts are local `@font-face` only; no network fetch during render. (spec §5)
- `scripts/forbidden.txt`, `.superpowers/`, `build/` stay git-ignored; `scripts/forbidden.example.txt` is committed. (spec §3)
- Post markdown, prompt files, and the check script's rules do not change. (spec §2)
- Banner URL pattern: `https://jenova-marie.github.io/the-client-seat/assets/banners/<slug>.png` where `<slug>` is the post filename without `.md`. (spec §6)
- Part titles on banners come from the post H1 after `Part N: `; part 0's H1 is unchanged. (spec §4, §6)
- The folder rename is the last step; nothing runs after it in this session. (spec §4)
- Commit after every task; push at the end of Task 5. The initial commit and the spec are already on `origin/main`.

```python
#!/usr/bin/env python3
"""Verify the banner manifest and PNGs. Exit 0 on pass, 1 on fail."""
import json
import re
import sys
from pathlib import Path

ROOT = Path(__file__).resolve().parent.parent
MANIFEST = ROOT / "assets" / "banners.json"
H1 = re.compile(r"^# Part (\d+): (.+?)\s*$", re.M)


def post_titles() -> dict[int, str]:
    titles = {}
    for post in sorted(ROOT.glob("[01]*.md")):
        m = H1.search(post.read_text())
        if m:
            titles[int(m.group(1))] = m.group(2)
    return titles


def main() -> int:
    fails = []
    if not MANIFEST.exists():
        print("FAIL: assets/banners.json missing")
        return 1
    entries = json.loads(MANIFEST.read_text())
    titles = post_titles()
    if len(entries) != 12:
        fails.append(f"manifest has {len(entries)} entries, expected 12")
    for e in entries:
        want = titles.get(e["part"])
        if want != e["title"]:
            fails.append(f"part {e['part']}: manifest title {e['title']!r} != post H1 {want!r}")
        if not e["slug"] or not e["file"].endswith(f"{e['slug']}.png"):
            fails.append(f"part {e['part']}: slug/file mismatch")
    for f in fails:
        print("FAIL:", f)
    print("PASS banners" if not fails else "FAIL banners")
    return 1 if fails else 0


if __name__ == "__main__":
    sys.exit(main())
```

```python
#!/usr/bin/env python3
"""Build the banner manifest from the post H1s and render one PNG per part."""
import json
import re
import sys
from pathlib import Path

ROOT = Path(__file__).resolve().parent.parent
ASSETS = ROOT / "assets"
MANIFEST = ASSETS / "banners.json"
H1 = re.compile(r"^# Part (\d+): (.+?)\s*$", re.M)


def read_parts() -> list[dict]:
    parts = []
    for post in sorted(ROOT.glob("[01]*.md")):
        m = H1.search(post.read_text())
        if not m:
            sys.exit(f"no 'Part N: title' H1 in {post.name}")
        slug = post.stem
        parts.append({
            "part": int(m.group(1)),
            "title": m.group(2),
            "slug": slug,
            "file": f"assets/banners/{slug}.png",
        })
    return parts


def write_manifest(parts: list[dict]) -> Path:
    MANIFEST.write_text(json.dumps(parts, indent=2) + "\n")
    return MANIFEST


def main() -> int:
    parts = read_parts()
    write_manifest(parts)
    print(f"manifest: {len(parts)} parts -> {MANIFEST.relative_to(ROOT)}")
    return 0


if __name__ == "__main__":
    sys.exit(main())
```

```python
def fill(template: str, part: dict, bg_tag: str) -> str:
    title = html.escape(part["title"], quote=True)
    return (template
            .replace("{{SERIES}}", SERIES)
            .replace("{{PART}}", str(part["part"]))
            .replace("{{TITLE}}", title)
            .replace("{{TITLE_CLASS}}", "long" if len(part["title"]) > 30 else "short")
            .replace("{{AUTHOR}}", AUTHOR)
            .replace("{{HUE}}", str((part["part"] * 9) % 360))
            .replace("{{BG}}", bg_tag))


def render(parts: list[dict]) -> None:
    if not os.path.exists(CHROME):
        sys.exit(f"Chrome not found at {CHROME}; set CHROME_BIN to your Chrome binary")
    BUILD.mkdir(parents=True, exist_ok=True)
    OUT.mkdir(parents=True, exist_ok=True)
    if BACKGROUND.exists():
        bg_tag = '<img class="bg" src="../../assets/series-background.png" alt="">'
    else:
        bg_tag = '<div class="bg-fallback"></div>'
        print("notice: assets/series-background.png not found; rendering the gradient fallback")
    template = TEMPLATE.read_text().replace('url("fonts/', 'url("../../assets/fonts/')
    for part in parts:
        page = BUILD / f"{part['slug']}.html"
        page.write_text(fill(template, part, bg_tag))
        png = ROOT / part["file"]
        cmd = [CHROME, "--headless=new", "--disable-gpu", "--hide-scrollbars",
               "--force-device-scale-factor=1", "--window-size=1600,900",
               "--virtual-time-budget=3000", f"--screenshot={png}", page.resolve().as_uri()]
        result = subprocess.run(cmd, capture_output=True, text=True)
        if result.returncode != 0 or not png.exists():
            sys.exit(f"screenshot failed for part {part['part']}: {result.stderr.strip()[:300]}")
        print(f"rendered {part['file']}")
```

```bash
PY
```

- [ ] **Step 1: Generate the index page from the manifest**

```bash
python3 - <<'PY'
import json, html
parts = json.load(open("assets/banners.json"))
base = "https://jenova-marie.github.io/the-client-seat/assets/banners"
items = "\n".join(
    f'    <li><a href="{base}/{p["slug"]}.png"><img src="assets/banners/{p["slug"]}.png" alt="Part {p["part"]}: {html.escape(p["title"])}" width="400"></a><br>'
    f'Part {p["part"]}: {html.escape(p["title"])}<br><code>{base}/{p["slug"]}.png</code></li>'
    for p in parts)
page = f"""<!doctype html>
<html lang="en"><head><meta charset="utf-8"><meta name="viewport" content="width=device-width, initial-scale=1">
<title>The Client Seat</title>
<style>body{{font-family:system-ui,sans-serif;max-width:900px;margin:40px auto;padding:0 16px;line-height:1.5}}
li{{margin:24px 0;list-style:none}}img{{display:block;max-width:100%;border-radius:12px}}code{{font-size:.85em}}</style>
</head><body>
<h1>The Client Seat</h1>
<p>Learning a real software project with AI as your tutor. The series is published on Medium; this page hosts the banner images.</p>
<ul>
{items}
</ul>
</body></html>
"""
open("index.html", "w").write(page)
print("index.html written")
PY
grep -c "<li>" index.html
```
